cmake_minimum_required(VERSION 3.24)
project(MaidionisMusicResearch VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Torch 2.5.1 EXACT REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
get_filename_component(MUSIC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
if(NOT DEFINED MAIDIONIS_ROOT OR NOT EXISTS "${MAIDIONIS_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "Set MAIDIONIS_ROOT to the pinned external Maidionis checkout")
endif()
set(MAIDIONIS_NUMERICAL ON CACHE BOOL "Use Maidionis native numerical training" FORCE)
add_subdirectory("${MAIDIONIS_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/maidionis")

file(GLOB MUSIC_PYTHON_SOURCES CONFIGURE_DEPENDS RELATIVE "${MUSIC_ROOT}" "${MUSIC_ROOT}/*.py")
file(GLOB MUSIC_NATIVE_SOURCES CONFIGURE_DEPENDS RELATIVE "${MUSIC_ROOT}"
  "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.h")
set(MUSIC_SOURCES ${MUSIC_PYTHON_SOURCES} ${MUSIC_NATIVE_SOURCES} "native/CMakeLists.txt")
list(FILTER MUSIC_SOURCES EXCLUDE REGEX "^(data|results|build|runs|__pycache__)/")
list(SORT MUSIC_SOURCES)
set(MUSIC_SOURCE_INPUT "maidionis-music-sources-v1\n")
foreach(source IN LISTS MUSIC_SOURCES)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${MUSIC_ROOT}/${source}")
  file(SHA256 "${MUSIC_ROOT}/${source}" source_digest)
  string(APPEND MUSIC_SOURCE_INPUT "${source}\n${source_digest}\n")
endforeach()
string(SHA256 MUSIC_SOURCE_DIGEST "${MUSIC_SOURCE_INPUT}")
add_library(maidionis_music_composition STATIC composition.cpp)
target_link_libraries(maidionis_music_composition PUBLIC maidionis_core)
target_include_directories(maidionis_music_composition PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(maidionis_music_composition PRIVATE MUSIC_SOURCE_DIGEST="${MUSIC_SOURCE_DIGEST}")
add_executable(maidionis_music_driver driver.cpp)
target_link_libraries(maidionis_music_driver PRIVATE maidionis_music_composition)
