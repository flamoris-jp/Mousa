cmake_minimum_required(VERSION 3.24)
project(MousaScaleGeneration LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Torch 2.5.1 EXACT REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
add_subdirectory(${MAIDIONIS_SOURCE} core)
add_executable(generate_scale_checkpoint generate.cpp)
target_link_libraries(generate_scale_checkpoint PRIVATE maidionis_core)
add_executable(scale_rule_tests scale_rule_tests.cpp)
add_executable(degree_model ../maidionis-degree-generation/degree_model.cpp)
target_link_libraries(degree_model PRIVATE maidionis_core)
